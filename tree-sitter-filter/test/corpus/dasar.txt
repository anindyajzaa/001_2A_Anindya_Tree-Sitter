==================

status = "open"

(query
  (comparison

==================
perbandingan dengan angka
==================

priority >= 2

---

(query
  (comparison
    (identifier)
    (number)))

==================
and menggabungkan dua perbandingan
==================

a = 1 and b = 2

---

(query
  (and_expression
    (comparison
      (identifier)
      (number))
    (comparison
      (identifier)
      (number))))

==================
and lebih kuat dari or (and di kanan)
==================

a = 1 or b = 2 and c = 3

---

(query
  (or_expression
    (comparison
      (identifier)
      (number))
    (and_expression
      (comparison
        (identifier)
        (number))
      (comparison
        (identifier)
        (number)))))

==================
and lebih kuat dari or (and di kiri)
==================

a = 1 and b = 2 or c = 3

---

(query
  (or_expression
    (and_expression
      (comparison
        (identifier)
        (number))
      (comparison
        (identifier)
        (number)))
    (comparison
      (identifier)
      (number))))

==================
kurung mendahulukan or
==================

(a = 1 or b = 2) and c = 3

---

(query
  (and_expression
    (parenthesized
      (or_expression
        (comparison
          (identifier)
          (number))
        (comparison
          (identifier)
          (number))))
    (comparison
      (identifier)
      (number))))
